local function detectDevice()
	local ok, name = pcall(function()
		local cam = workspace.CurrentCamera
		local vp = cam and cam.ViewportSize or Vector2.new(1280, 720)
		local touch = UserInputService.TouchEnabled
		local keyboard = UserInputService.KeyboardEnabled
		if touch and not keyboard then
			if math.min(vp.X, vp.Y) >= 600 then
				return "Tablet"
			end
			return "Phone"
		end
		return "PC"
	end)
	return ok and name or "PC"
end
SpaceHub.DetectDevice = detectDevice

local function lum(c)
	return 0.2126 * c.R + 0.7152 * c.G + 0.0722 * c.B
end
local function ensureVisible(c, bg, minDiff)
	local target = lum(bg) < 0.5 and Color3.new(1, 1, 1) or Color3.fromRGB(12, 14, 20)
	for _ = 1, 12 do
		if math.abs(lum(c) - lum(bg)) >= minDiff then
			break
		end
		c = c:Lerp(target, 0.15)
	end
	return c
end

	t.Text = t.Text or ensureVisible(t.Accent:Lerp(extreme, 0.82), t.Background, 0.6)
	t.Muted = t.Muted or ensureVisible(t.Accent:Lerp(extreme, 0.2), t.Background, 0.4)
	t.Flash = t.Flash or ensureVisible(t.Accent:Lerp(extreme, 0.15), t.Background, 0.35)
end

local function tw(obj, t, props, style, dir)
	local tween = TweenService:Create(obj, TweenInfo.new(t, style or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out), props)
	tween:Play()
	return tween
end

local function apply(obj, props, animate, t, style)
	if animate then
		return tw(obj, t or 0.3, props, style)
	end
	for k, v in pairs(props) do
		obj[k] = v
	end
end

local SOFT_PROPS = { "TextColor3", "PlaceholderColor3", "ImageColor3" }
local function new(class, props, parent)
	local o = Instance.new(class)
	for k, v in pairs(props) do
		o[k] = v
	end
	o.Parent = parent
	for _, prop in ipairs(SOFT_PROPS) do
		local v = props[prop]
		if v ~= nil and (v == TEXT or v == MUTED) then
			table.insert(softRegistry, { o, prop })
		end
	end
	return o
end

local function corner(obj, r)
	return new("UICorner", { CornerRadius = typeof(r) == "UDim" and r or UDim.new(0, r) }, obj)
end

local function pad(obj, top, right, bottom, left)
	return new("UIPadding", {
		PaddingTop = UDim.new(0, top), PaddingRight = UDim.new(0, right),
		PaddingBottom = UDim.new(0, bottom), PaddingLeft = UDim.new(0, left),
	}, obj)
end

local function themed(obj, prop, key)
	obj[prop] = Theme[key]
	table.insert(themeRegistry, { obj, prop, key })
	return obj
end

local function applyTheme(t)
	local oldText, oldMuted = TEXT, MUTED
	Theme = t
	TEXT, MUTED = t.Text, t.Muted
	for i = #softRegistry, 1, -1 do
		local r = softRegistry[i]
		local inst, prop = r[1], r[2]
		if inst.Parent then
			local cur = inst[prop]
			if cur == oldText then
				tw(inst, 0.35, { [prop] = TEXT })
			elseif cur == oldMuted then
				tw(inst, 0.35, { [prop] = MUTED })
			end
		else
			table.remove(softRegistry, i)
		end
	end
	for i = #themeRegistry, 1, -1 do
		local r = themeRegistry[i]
		if r[1].Parent then
			tw(r[1], 0.35, { [r[2]] = t[r[3]] })
		else
			table.remove(themeRegistry, i)
		end
	end
	for _, hook in ipairs(themeHooks) do
		task.spawn(hook)
	end
end

local function setThemeByName(name)
	for _, t in ipairs(THEMES) do
		if t.Name:lower() == tostring(name):lower() then
			applyTheme(t)
			local flag = SpaceHub.Flags["SpaceHub_Theme"]
			if flag then
				flag.CurrentValue = t.Name
			end
			return true
		end
	end
	return false
end

local HAPTIC_STRENGTH = 0.22
local HAPTIC_TIME = 0.05
local hapticBusy = false
local function haptic()
	if not SpaceHub.Haptics or hapticBusy then
		return
	end
	hapticBusy = true
	task.spawn(function()
		local fired = false
		for _, inputType in ipairs({ Enum.UserInputType.Gamepad1, Enum.UserInputType.Touch }) do
			pcall(function()
				if HapticService:IsVibrationSupported(inputType) and HapticService:IsMotorSupported(inputType, Enum.VibrationMotor.Small) then
					HapticService:SetMotor(inputType, Enum.VibrationMotor.Small, HAPTIC_STRENGTH)
					fired = true
					task.wait(HAPTIC_TIME)
					HapticService:SetMotor(inputType, Enum.VibrationMotor.Small, 0)
				end
			end)
			if fired then
				break
			end
		end
		task.wait(0.08)
		hapticBusy = false
	end)
end
SpaceHub.Haptic = haptic

local function overlayFlash(target, radius, color, alpha, duration, withStroke)
	if not (target and target.Parent) then
		return
	end
	local fl = new("Frame", {
		Size = UDim2.fromScale(1, 1), BackgroundColor3 = color, BackgroundTransparency = alpha,
		BorderSizePixel = 0, ZIndex = 20,
	}, target)
	corner(fl, radius or 8)
	local d = duration or 0.55
	if withStroke then
		local st = new("UIStroke", { Color = color, Thickness = 2, Transparency = 0.05 }, fl)
		tw(st, d, { Transparency = 1 })
	end
	tw(fl, d, { BackgroundTransparency = 1 }).Completed:Connect(function()
		fl:Destroy()
	end)
end

local function glowFlash(target, color, duration)
	local host = target and target.Parent
	if not host then
		return
	end
	color = color or Theme.Flash
	duration = duration or 0.75
	local ax, ay = target.AnchorPoint.X, target.AnchorPoint.Y
	local layers = {
		{ pad = 3, alpha = 0.45 },
		{ pad = 7, alpha = 0.66 },
		{ pad = 12, alpha = 0.82 },
	}
	for i, L in ipairs(layers) do
		local pad = L.pad
		local g = new("Frame", {
			AnchorPoint = target.AnchorPoint,
			Position = target.Position + UDim2.fromOffset(pad * (2 * ax - 1), pad * (2 * ay - 1)),
			Size = target.Size + UDim2.fromOffset(pad * 2, pad * 2),
			BackgroundColor3 = color, BackgroundTransparency = L.alpha, BorderSizePixel = 0,
			ZIndex = target.ZIndex + 18 + i, Active = false,
		}, host)
		corner(g, 6 + pad)
		local grow = pad + 4
		local t = tw(g, duration, {
			BackgroundTransparency = 1,
			Position = target.Position + UDim2.fromOffset(grow * (2 * ax - 1), grow * (2 * ay - 1)),
			Size = target.Size + UDim2.fromOffset(grow * 2, grow * 2),
		})
		t.Completed:Connect(function()
			g:Destroy()
		end)
	end
end

local function flashGlow(target, radius, alpha)

end

local function safeTap(hit, scroller, onTap)
	local press
	local lastScroll = 0
	if scroller then
		scroller:GetPropertyChangedSignal("CanvasPosition"):Connect(function()
			lastScroll = os.clock()
			if press then
				press.scrolled = true
			end
		end)
	end
	hit.InputBegan:Connect(function(input)
		local t = input.UserInputType
		if t == Enum.UserInputType.MouseButton1 or t == Enum.UserInputType.Touch then
			press = {
				pos = input.Position, time = os.clock(), scrolled = false,
				settled = (os.clock() - lastScroll) > 0.15, kind = t,
			}
		end
	end)
	bind(UserInputService.InputEnded, function(input)
		if not press or input.UserInputType ~= press.kind then
			return
		end
		local p = press
		press = nil
		local moved = (input.Position - p.pos).Magnitude
		if p.settled and not p.scrolled and moved <= 10 and (os.clock() - p.time) <= 0.8 then
			onTap()
		end
	end)
end

local function tintIcon(obj, color, animate)
	if not (obj and obj.Parent) then
		return
	end
	local function one(o)
		local props
		if o:IsA("ImageLabel") or o:IsA("ImageButton") then
			if o.Image ~= "" then
				props = { ImageColor3 = color }
			end
		elseif o:IsA("TextLabel") then
			props = { TextColor3 = color }
		elseif o:IsA("UIStroke") then
			props = { Color = color }
		elseif o:IsA("Frame") and o.BackgroundTransparency < 1 then
			props = { BackgroundColor3 = color }
		end
		if props then
			apply(o, props, animate, 0.15)
		end
	end
	one(obj)
	for _, d in ipairs(obj:GetDescendants()) do
		one(d)
	end
end

local function iconPress(hit, icon)
	tintIcon(icon, ICON_IDLE, false)
	local kind
	hit.InputBegan:Connect(function(input)
		local t = input.UserInputType
		if t == Enum.UserInputType.MouseButton1 or t == Enum.UserInputType.Touch then
			kind = t
			tintIcon(icon, WHITE, true)
		end
	end)
	bind(UserInputService.InputEnded, function(input)
		if kind and input.UserInputType == kind then
			kind = nil
			tintIcon(icon, ICON_IDLE, true)
		end
	end)
end

local function normalize(s)
	s = tostring(s)
	for _, p in ipairs(ACCENTS) do
		s = s:gsub(p[1], p[2])
	end
	return (s:lower():gsub("[^%w]", ""))
end

local function safe(fn, ...)
	if type(fn) ~= "function" then
		return
	end
	local ok, err = pcall(fn, ...)
	if not ok then
		warn("[Space Hub] Callback error: " .. tostring(err))
		SpaceHub:Notify({ Title = "Callback error", Content = tostring(err), Duration = 6 })
	end
end

local saveQueued = false
local function autosave()
	local s = SpaceHub._settings
	if not (s and s.ConfigurationSaving and s.ConfigurationSaving.Enabled) or not SpaceHub.AutoSave then
		return
	end
	if saveQueued then
		return
	end
	saveQueued = true
	task.delay(1, function()
		saveQueued = false
		SpaceHub:SaveConfiguration()
	end)
end

local function applyIcon(img, icon)
	local spec
	if type(icon) == "number" then
		spec = { Image = "rbxassetid://" .. icon }
	elseif type(icon) == "string" and icon ~= "" then
		if icon:find("rbxasset") or icon:find("^https?://") then
			spec = { Image = icon }
		elseif icon:match("^%d+$") then
			spec = { Image = "rbxassetid://" .. icon }
		else
			local v = SpaceHub.Icons[icon:lower()]
			if type(v) == "string" then
				spec = { Image = v }
			elseif type(v) == "number" then
				spec = { Image = "rbxassetid://" .. v }
			elseif type(v) == "table" then
				spec = v
			end
		end
	end
	if not spec then
		return false
	end
	img.Image = spec.Image
	if spec.ImageRectSize then
		img.ImageRectSize = spec.ImageRectSize
	end
	if spec.ImageRectOffset then
		img.ImageRectOffset = spec.ImageRectOffset
	end
	return true
end

function SpaceHub:SetIcons(source)
	local map = source
	if type(source) == "string" then
		local ok, res = pcall(function()
			return loadstring(game:HttpGet(source))()
		end)
		map = ok and res or nil
	end
	if type(map) == "table" then
		for k, v in pairs(map) do
			self.Icons[tostring(k):lower()] = v
		end
	end
end

local function loadImage(obj, id, onFail)
	if not id or id == "" then
		onFail()
		return
	end
	task.spawn(function()
		local ok = false
		pcall(function()
			ContentProvider:PreloadAsync({ obj }, function(_, status)
				ok = (status == Enum.AssetFetchStatus.Success)
			end)
		end)
		if not ok then
			onFail()
		end
	end)
end

local function makeLabel(parent, props)
	props.BackgroundTransparency = 1
	props.Font = props.Font or Enum.Font.Gotham
	props.TextSize = props.TextSize or 13
	props.TextColor3 = props.TextColor3 or TEXT
	props.TextXAlignment = props.TextXAlignment or Enum.TextXAlignment.Left
	return new("TextLabel", props, parent)
end

local function makeButton(parent, caption, order, height, c1, c2, left)
	local frame = new("Frame", {
		LayoutOrder = order, Size = UDim2.new(1, 0, 0, height), BackgroundColor3 = WHITE, BorderSizePixel = 0,
	}, parent)
	corner(frame, 8)
	new("UIGradient", { Rotation = 90, Color = ColorSequence.new(c1 or STEEL_A, c2 or STEEL_B) }, frame)
	metalStroke(frame, 2)
	local button = new("TextButton", {
		Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, AutoButtonColor = false,
		Text = caption, Font = Enum.Font.GothamMedium, TextSize = 14, TextColor3 = TEXT,
		TextXAlignment = left and Enum.TextXAlignment.Left or Enum.TextXAlignment.Center,
	}, frame)
	if left then
		new("UIPadding", { PaddingLeft = UDim.new(0, 14) }, button)
	end
	return frame, button
end

local function expandable(parent, order, name, bodyHeight)
	local cont = new("Frame", {
		LayoutOrder = order, Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y, BackgroundTransparency = 1,
	}, parent)
	new("UIListLayout", { Padding = UDim.new(0, 6), SortOrder = Enum.SortOrder.LayoutOrder }, cont)
	local head = row(cont, 40, 1)
	local nameLbl = makeLabel(head, {
		Position = UDim2.fromOffset(12, 0), Size = UDim2.new(0.55, -12, 1, 0),
		Text = name, Font = Enum.Font.GothamMedium, TextSize = 14,
	})
	local arrow = new("ImageLabel", {
		AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -12, 0.5, 0), Size = UDim2.fromOffset(18, 18),
		BackgroundTransparency = 1, Image = ASSETS.Arrow, ImageColor3 = ICON_IDLE, ScaleType = Enum.ScaleType.Fit,
	}, head)
	loadImage(arrow, ASSETS.Arrow, function()
		arrow.Image = ""
		makeLabel(arrow, { Size = UDim2.fromScale(1, 1), Text = "v", Font = Enum.Font.GothamBold, TextXAlignment = Enum.TextXAlignment.Center })
		tintIcon(arrow, ICON_IDLE, false)
	end)
	local body = new("Frame", {
		LayoutOrder = 2, Size = UDim2.new(1, 0, 0, 0), BackgroundTransparency = 1, ClipsDescendants = true,
	}, cont)
	local open = false
	local ex = { Container = cont, Head = head, Name = nameLbl, Arrow = arrow, Body = body }
	function ex.SetOpen(v, h)
		open = v
		tw(body, 0.3, { Size = UDim2.new(1, 0, 0, v and (h or bodyHeight) or 0) })
		tw(arrow, 0.3, { Rotation = v and 180 or 0 })
	end
	function ex.IsOpen()
		return open
	end
	local click = new("TextButton", { Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, Text = "", ZIndex = 2 }, head)
	iconPress(click, arrow)
	click.Activated:Connect(function()
		ex.SetOpen(not open)
	end)
	return ex
end

local function guiParent()
	local ok, core = pcall(function()

	end)

		local writable = pcall(function()

		end)

		end
	end

local function keepOnTop(gui)
	local cont = gui.Parent
	local last = 0
	local con = cont.ChildAdded:Connect(function(child)
		if child ~= gui and child:IsA("ScreenGui") and os.clock() - last > 1 then
			last = os.clock()
			task.defer(function()
				if gui.Parent == cont then
					gui.Parent = nil
					gui.Parent = cont
				end
			end)
		end
	end)
	gui.Destroying:Connect(function()
		con:Disconnect()
	end)
end

		DisplayOrder = 2147483647, ZIndexBehavior = Enum.ZIndexBehavior.Sibling,
	}, parent)
	keepOnTop(gui)
	table.insert(allGuis, gui)
	return gui
end

local function configPath()
	local cs = (SpaceHub._settings and SpaceHub._settings.ConfigurationSaving) or {}
	local folder = cs.FolderName or "SpaceHub"
	return folder, folder .. "/" .. (cs.FileName or "config") .. ".json"
end

function SpaceHub:SaveConfiguration()
	if not canFile() then
		return false
	end
	local data = {}
	for flag, el in pairs(self.Flags) do
		local t = el.Type
		if t == "ColorPicker" then
			local c = el.Color
			data[flag] = { R = c.R, G = c.G, B = c.B }
		elseif t == "Dropdown" then
			data[flag] = el.CurrentOption
		elseif t == "Keybind" then
			data[flag] = el.CurrentKeybind
		else
			data[flag] = el.CurrentValue
		end
	end
	local folder, path = configPath()
	local ok = pcall(function()
		if type(isfolder) == "function" and type(makefolder) == "function" and not isfolder(folder) then
			makefolder(folder)
		end
		writefile(path, HttpService:JSONEncode(data))
	end)
	return ok
end

function SpaceHub:LoadConfiguration()
	if not canFile() then
		return false
	end
	local _, path = configPath()
	local okFile, exists = pcall(isfile, path)
	if not okFile or not exists then
		return false
	end
	local ok, data = pcall(function()
		return HttpService:JSONDecode(readfile(path))
	end)
	if not ok or type(data) ~= "table" then
		return false
	end
	for flag, value in pairs(data) do
		local el = self.Flags[flag]
		if el then
			pcall(function()
				if el.Type == "ColorPicker" then
					el:Set(Color3.new(value.R, value.G, value.B))
				else
					el:Set(value)
				end
			end)
		end
	end
	return true
end

local RED_A, RED_B = C(205, 60, 60), C(140, 32, 40)
local GRAY_A, GRAY_B = C(92, 95, 106), C(60, 62, 72)

local function runKeySystem(settings)
	local ks = settings.KeySettings or {}
	local keys = ks.Key or {}
	if type(keys) == "string" then
		keys = { keys }
	end
	local valid = {}
	for _, k in ipairs(keys) do
		if ks.GrabKeyFromSite then
			local ok, res = pcall(function()
				return game:HttpGet(k)
			end)
			if ok and type(res) == "string" then
				table.insert(valid, (res:gsub("%s", "")))
			end
		else
			table.insert(valid, tostring(k))
		end
	end
	local function check(txt)
		txt = (tostring(txt or ""):gsub("^%s+", ""):gsub("%s+$", ""))
		for _, k in ipairs(valid) do
			if txt == k then
				return true
			end
		end
		return false
	end

	local folder = (settings.ConfigurationSaving and settings.ConfigurationSaving.FolderName) or "SpaceHub"
	local path = folder .. "/" .. (ks.FileName or "Key") .. ".txt"
	if ks.SaveKey ~= false and canFile() then
		local ok, saved = pcall(function()
			return isfile(path) and readfile(path) or nil
		end)
		if ok and saved and check(saved) then
			return true
		end
	end

	local gui = createGui("SpaceHubKey")
	local modal = makeModal(gui, 360)
	local card = modal.Card
	modalText(card, 1, ks.Title or "Key System", 17, TEXT, Enum.Font.GothamBold)
	if ks.Subtitle then
		modalText(card, 2, ks.Subtitle, 13, MUTED)
	end
	if ks.Note then
		modalText(card, 3, ks.Note, 12, MUTED)
	end
	local boxFrame = new("Frame", { LayoutOrder = 4, Size = UDim2.new(1, 0, 0, 38), BackgroundColor3 = C(20, 22, 28), BorderSizePixel = 0 }, card)
	corner(boxFrame, 8)
	metalStroke(boxFrame, 1.5)
	local box = new("TextBox", {
		Position = UDim2.fromOffset(10, 0), Size = UDim2.new(1, -20, 1, 0), BackgroundTransparency = 1, ClearTextOnFocus = false,
		PlaceholderText = "Enter key...", PlaceholderColor3 = MUTED, Text = "", Font = Enum.Font.Gotham,
		TextSize = 14, TextColor3 = TEXT, TextXAlignment = Enum.TextXAlignment.Left,
	}, boxFrame)
	local msg = modalText(card, 5, "", 12, RED)
	local defs = { { Text = "Submit" }, { Text = "Cancel", C1 = GRAY_A, C2 = GRAY_B } }
	if ks.KeyLink then
		table.insert(defs, 2, { Text = "Get key" })
	end
	local buttons = buttonRow(card, 6, defs)

	local result
	local function submit()
		if check(box.Text) then
			if ks.SaveKey ~= false and canFile() then
				pcall(function()
					if type(isfolder) == "function" and type(makefolder) == "function" and not isfolder(folder) then
						makefolder(folder)
					end
					writefile(path, box.Text)
				end)
			end
			result = true
		else
			msg.Text = "Invalid key. Try again."
			for i = 1, 6 do
				card.Position = UDim2.new(0.5, (i % 2 == 0) and 8 or -8, 0.5, 0)
				task.wait(0.04)
			end
			card.Position = UDim2.fromScale(0.5, 0.5)
		end
	end
	buttons[1].Activated:Connect(submit)
	box.FocusLost:Connect(function(enter)
		if enter then
			submit()
		end
	end)
	if ks.KeyLink then
		buttons[2].Activated:Connect(function()
			msg.Text = copyText(ks.KeyLink) and "Link copied to clipboard." or ks.KeyLink
			msg.TextColor3 = MUTED
		end)
	end
	buttons[#buttons].Activated:Connect(function()
		result = false
	end)
	repeat
		task.wait(0.1)
	until result ~= nil

	local closed = false
	modal.Close(function()
		closed = true
	end)
	repeat
		task.wait(0.05)
	until closed
	gui:Destroy()
	if not result then
		error("[Space Hub] Key system closed.", 0)
	end
	return true
end

local function discordPrompt(d)
	if not d or not d.Enabled or not d.Invite or d.Invite == "" then
		return
	end
	local folder = "SpaceHub"
	local mark = folder .. "/discord_" .. tostring(d.Invite):gsub("[^%w]", "") .. ".txt"
	if d.RememberJoins ~= false and canFile() then
		local ok, seen = pcall(isfile, mark)
		if ok and seen then
			return
		end
	end
	local gui = createGui("SpaceHubDiscord")
	local modal = makeModal(gui, 340)
	modalText(modal.Card, 1, d.Title or "Join our Discord", 17, TEXT, Enum.Font.GothamBold)
	modalText(modal.Card, 2, d.Content or "Get updates, support and more.", 13, MUTED)
	modalText(modal.Card, 3, "discord.gg/" .. d.Invite, 14, MUTED, Enum.Font.GothamMedium)
	local buttons = buttonRow(modal.Card, 4, { { Text = "Copy invite" }, { Text = "Not now", C1 = GRAY_A, C2 = GRAY_B } })
	local function done(copied)
		modal.Close(function()
			gui:Destroy()
		end)
		if copied and d.RememberJoins ~= false and canFile() then
			pcall(function()
				if type(isfolder) == "function" and type(makefolder) == "function" and not isfolder(folder) then
					makefolder(folder)
				end
				writefile(mark, "1")
			end)
		end
	end
	buttons[1].Activated:Connect(function()
		local ok = copyText("https://discord.gg/" .. d.Invite)
		SpaceHub:Notify({ Title = ok and "Invite copied" or "Copy not supported", Content = "discord.gg/" .. d.Invite, Duration = 4 })
		done(ok)
	end)
	buttons[2].Activated:Connect(function()
		done(false)
	end)
end

local function makeApi(page, tab, register)
	local api = {}
	local order = 0
	local function nextOrder()
		order += 1
		return order
	end
	api._nextOrder = nextOrder

	local function flag(o, obj)
		if o.Flag then
			SpaceHub.Flags[o.Flag] = obj
		end
	end

	function api:CreateSection(name)
		local f = new("Frame", { LayoutOrder = nextOrder(), Size = UDim2.new(1, 0, 0, 26), BackgroundTransparency = 1 }, page)
		local t = makeLabel(f, { Size = UDim2.new(1, 0, 1, -4), Text = name, Font = Enum.Font.GothamBold, TextColor3 = MUTED, TextSize = 12 })
		local line = new("Frame", { AnchorPoint = Vector2.new(0, 1), Position = UDim2.new(0, 0, 1, 0), Size = UDim2.new(1, 0, 0, 1), BorderSizePixel = 0 }, f)
		themed(line, "BackgroundColor3", "Line")
		local obj = { Type = "Section" }
		function obj:Set(n)
			t.Text = n
		end
		return obj
	end

	function api:CreateLabel(txt, icon)
		local f = row(page, 36, nextOrder())
		local x = 12
		if icon then
			local img = new("ImageLabel", { Position = UDim2.fromOffset(10, 9), Size = UDim2.fromOffset(18, 18), BackgroundTransparency = 1, ImageColor3 = WHITE }, f)
			if applyIcon(img, icon) then
				x = 36
			else
				img:Destroy()
			end
		end
		local l = makeLabel(f, { Position = UDim2.fromOffset(x, 0), Size = UDim2.new(1, -x - 12, 1, 0), Text = txt })
		register(txt, "Label", f, tab)
		local obj = { Type = "Label" }
		function obj:Set(t)
			l.Text = t
		end
		return obj
	end

	function api:CreateParagraph(o)
		local f = row(page, 0, nextOrder())
		f.AutomaticSize = Enum.AutomaticSize.Y
		pad(f, 10, 12, 10, 12)
		new("UIListLayout", { Padding = UDim.new(0, 4), SortOrder = Enum.SortOrder.LayoutOrder }, f)
		local title = makeLabel(f, {
			LayoutOrder = 1, Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y,
			Text = o.Title or "", Font = Enum.Font.GothamBold, TextSize = 14, TextWrapped = true,
		})
		local body = makeLabel(f, {
			LayoutOrder = 2, Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y,
			Text = o.Content or "", TextSize = 12, TextColor3 = MUTED, TextWrapped = true, TextYAlignment = Enum.TextYAlignment.Top,
		})
		register(o.Title or "Paragraph", "Paragraph", f, tab)
		local obj = { Type = "Paragraph" }
		function obj:Set(v)
			title.Text = v.Title or title.Text
			body.Text = v.Content or body.Text
		end
		return obj
	end

	function api:CreateButton(o)
		local f = row(page, 40, nextOrder())
		local b = new("TextButton", {
			Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, AutoButtonColor = false, Text = o.Name,
			Font = Enum.Font.GothamMedium, TextSize = 14, TextColor3 = TEXT,
		}, f)
		b.Activated:Connect(function()
			tw(b, 0.08, { TextTransparency = 0.5 }).Completed:Connect(function()
				tw(b, 0.15, { TextTransparency = 0 })
			end)
			safe(o.Callback)
		end)
		register(o.Name, "Button", f, tab)
		local obj = { Type = "Button" }
		function obj:Set(n)
			b.Text = n
		end
		return obj
	end

	function api:CreateToggle(o)
		local TRACK_ON = C(54, 226, 104)
		local TRACK_OFF = C(98, 101, 110)
		local STROKE_ON = C(176, 255, 204)
		local STROKE_OFF = C(122, 140, 130)
		local LAYERS = 200
		local GLOW_REACH = 56

		local f = row(page, 40, nextOrder())
		makeLabel(f, { Position = UDim2.fromOffset(12, 0), Size = UDim2.new(1, -80, 1, 0), Text = o.Name, Font = Enum.Font.GothamMedium, TextSize = 14 })
		local holder = new("Frame", {
			AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -12, 0.5, 0), Size = UDim2.fromOffset(46, 26), BackgroundTransparency = 1,
		}, f)

		local glowGroup = new("Frame", {
			AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.fromScale(0.5, 0.5),
			Size = UDim2.new(1, GLOW_REACH * 2 + 4, 1, GLOW_REACH * 2 + 4),
			BackgroundTransparency = 1, ZIndex = 1, Visible = false,
		}, holder)
		local glowLayers = {}
		local outerColor, innerColor = C(40, 235, 170), C(40, 235, 105)
		for i = 1, LAYERS do
			local k = i / LAYERS
			local extra = (LAYERS - i + 1) * (GLOW_REACH / LAYERS)
			local alpha = 0.0008 + 0.0165 * k * k
			local layer = new("Frame", {
				AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.fromScale(0.5, 0.5),
				Size = UDim2.new(0, 46 + extra * 2, 0, 26 + extra * 2),
				BackgroundColor3 = outerColor:Lerp(innerColor, k), BackgroundTransparency = 1,
				BorderSizePixel = 0, ZIndex = 1,
			}, glowGroup)
			corner(layer, UDim.new(1, 0))
			glowLayers[i] = { layer, alpha }
		end
		local glowValue = new("NumberValue", { Value = 0 }, glowGroup)
		glowValue.Changed:Connect(function(t)
			for i = 1, #glowLayers do
				local g = glowLayers[i]
				g[1].BackgroundTransparency = 1 - g[2] * t
			end
		end)

		local track = new("TextButton", {
			Size = UDim2.fromScale(1, 1), BackgroundColor3 = TRACK_OFF, BackgroundTransparency = 0,
			AutoButtonColor = false, Text = "", ZIndex = 3,
		}, holder)
		corner(track, UDim.new(1, 0))
		local trackStroke = new("UIStroke", { Thickness = 2, Color = STROKE_OFF, Transparency = 0.45, ApplyStrokeMode = Enum.ApplyStrokeMode.Border }, track)

	function api:CreateSlider(o)
		local min, max = o.Range[1], o.Range[2]
		local inc = o.Increment or 1
		local f = row(page, 54, nextOrder())
		makeLabel(f, { Position = UDim2.fromOffset(12, 6), Size = UDim2.new(0.6, -12, 0, 18), Text = o.Name, Font = Enum.Font.GothamMedium, TextSize = 14 })
		local valLbl = makeLabel(f, {
			AnchorPoint = Vector2.new(1, 0), Position = UDim2.new(1, -12, 0, 6), Size = UDim2.new(0.4, 0, 0, 18),
			TextXAlignment = Enum.TextXAlignment.Right, TextColor3 = MUTED, TextSize = 12,
		})
		local track = new("Frame", {
			Position = UDim2.new(0, 12, 0, 36), Size = UDim2.new(1, -24, 0, 6), BackgroundColor3 = C(58, 62, 76), BorderSizePixel = 0,
		}, f)
		corner(track, UDim.new(1, 0))
		local fill = new("Frame", { Size = UDim2.new(0, 0, 1, 0), BorderSizePixel = 0 }, track)
		themed(fill, "BackgroundColor3", "Accent")
		corner(fill, UDim.new(1, 0))
		local knob = new("Frame", {
			AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.new(0, 0, 0.5, 0), Size = UDim2.fromOffset(14, 14),
			BackgroundColor3 = WHITE, BorderSizePixel = 0, ZIndex = 2,
		}, track)
		corner(knob, UDim.new(1, 0))
		metalStroke(knob, 1.5)
		local hit = new("TextButton", {
			Position = UDim2.new(0, 0, 0, 26), Size = UDim2.new(1, 0, 0, 26), BackgroundTransparency = 1, Text = "",
		}, f)

		local obj = { Type = "Slider", CurrentValue = o.CurrentValue or min, Name = o.Name }
		local function round(v)
			v = math.clamp(v, min, max)
			v = math.floor((v - min) / inc + 0.5) * inc + min
			return math.clamp(math.floor(v * 10000 + 0.5) / 10000, min, max)
		end
		local function visual(v)
			local p = (max == min) and 0 or (v - min) / (max - min)
			fill.Size = UDim2.new(p, 0, 1, 0)
			knob.Position = UDim2.new(p, 0, 0.5, 0)
			valLbl.Text = tostring(v) .. (o.Suffix and (" " .. o.Suffix) or "")
		end
		obj.CurrentValue = round(obj.CurrentValue)
		visual(obj.CurrentValue)

		local dragging = false
		local function update(x)
			local rel = math.clamp((x - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X, 1), 0, 1)
			local v = round(min + rel * (max - min))
			if v ~= obj.CurrentValue then
				obj.CurrentValue = v
				visual(v)
				safe(o.Callback, v)
				autosave()
			end
		end
		hit.InputBegan:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
				dragging = true
				page.ScrollingEnabled = false
				update(input.Position.X)
			end
		end)
		bind(UserInputService.InputChanged, function(input)
			if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
				update(input.Position.X)
			end
		end)
		bind(UserInputService.InputEnded, function(input)
			if dragging and (input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch) then
				dragging = false
				page.ScrollingEnabled = true
			end
		end)
		function obj:Set(v)
			obj.CurrentValue = round(tonumber(v) or min)
			visual(obj.CurrentValue)
			safe(o.Callback, obj.CurrentValue)
			autosave()
		end
		register(o.Name, "Slider", f, tab)
		flag(o, obj)
		return obj
	end

	function api:CreateDropdown(o)
		local options = o.Options or {}
		local multiple = o.MultipleOptions == true
		local ex = expandable(page, nextOrder(), o.Name, 0)
		local valueLbl = makeLabel(ex.Head, {
			AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -38, 0.5, 0), Size = UDim2.new(0.4, 0, 0, 18),
			TextXAlignment = Enum.TextXAlignment.Right, TextColor3 = MUTED, TextSize = 12, TextTruncate = Enum.TextTruncate.AtEnd,
		})
		local list = new("ScrollingFrame", {
			Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, BorderSizePixel = 0, ScrollBarThickness = 3,
			CanvasSize = UDim2.new(), AutomaticCanvasSize = Enum.AutomaticSize.Y, ScrollingDirection = Enum.ScrollingDirection.Y,
		}, ex.Body)
		themed(list, "ScrollBarImageColor3", "Accent")
		new("UIListLayout", { Padding = UDim.new(0, 6), SortOrder = Enum.SortOrder.LayoutOrder }, list)
		pad(list, 3, 8, 3, 3)

		local selected = {}
		local buttons = {}
		local obj = { Type = "Dropdown", CurrentOption = {}, Name = o.Name }
		local function ordered()
			local r = {}
			for _, opt in ipairs(options) do
				if selected[opt] then
					table.insert(r, opt)
				end
			end
			return r
		end
		local function paint()
			for opt, b in pairs(buttons) do
				b.TextColor3 = selected[opt] and Theme.Accent or TEXT
			end
			local cur = ordered()
			valueLbl.Text = #cur > 0 and table.concat(cur, ", ") or "None"
		end
		local function commit(silent)
			obj.CurrentOption = ordered()
			paint()
			if not silent then
				safe(o.Callback, obj.CurrentOption)
				autosave()
			end
		end
		local function build()
			for _, b in pairs(buttons) do
				b.Parent:Destroy()
			end
			buttons = {}
			for i, opt in ipairs(options) do
				local _, b = makeButton(list, tostring(opt), i, 30)
				buttons[opt] = b
				b.Activated:Connect(function()
					if multiple then
						selected[opt] = not selected[opt] or nil
					else
						selected = { [opt] = true }
						ex.SetOpen(false)
					end
					commit(false)
				end)
			end
			local h = math.min(#options * 36 + 8, 182)
			if ex.IsOpen() then
				ex.SetOpen(true, h)
			end
			ex.Height = h
			paint()
		end
		local origSetOpen = ex.SetOpen
		ex.SetOpen = function(v, h)
			origSetOpen(v, h or ex.Height)
		end

		local initial = o.CurrentOption or {}
		if type(initial) == "string" then
			initial = { initial }
		end
		for _, v in ipairs(initial) do
			selected[v] = true
		end
		build()
		commit(true)

		function obj:Set(v)
			selected = {}
			if type(v) == "string" then
				v = { v }
			end
			for _, x in ipairs(v or {}) do
				selected[x] = true
			end
			commit(false)
		end
		function obj:Refresh(newOptions)
			options = newOptions or {}
			for k in pairs(selected) do
				if not table.find(options, k) then
					selected[k] = nil
				end
			end
			build()
			commit(true)
		end
		table.insert(themeHooks, paint)
		register(o.Name, "Dropdown", ex.Head, tab)
		flag(o, obj)
		return obj
	end

			CanvasSize = UDim2.new(), AutomaticCanvasSize = Enum.AutomaticSize.Y,
			ScrollingDirection = Enum.ScrollingDirection.Y,
		}, box)

			Size = UDim2.new(1, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.Y, BackgroundTransparency = 1,
			ClearTextOnFocus = false, MultiLine = true, TextWrapped = true,
			PlaceholderText = o.PlaceholderText or "Write your code here...", PlaceholderColor3 = MUTED,
			Text = o.CurrentValue or "", Font = Enum.Font.Code, TextSize = 12, TextColor3 = TEXT,
			TextXAlignment = Enum.TextXAlignment.Left, TextYAlignment = Enum.TextYAlignment.Top,
		}, scroll)
		pad(tb, 6, 8, 6, 8)
		new("UISizeConstraint", { MinSize = Vector2.new(0, math.max(height - 44, 20)) }, tb)

		local obj = { Type = "CodeInput", CurrentValue = o.CurrentValue or "", Name = o.Name or "Code input" }
		local lastText, lastCursor = tb.Text, #tb.Text + 1
		local reopening = false
		tb:GetPropertyChangedSignal("Text"):Connect(function()
			if tb:IsFocused() then
				lastText = tb.Text
			end
		end)
		tb:GetPropertyChangedSignal("CursorPosition"):Connect(function()
			if tb:IsFocused() and tb.CursorPosition > 0 then
				lastCursor = tb.CursorPosition
			end

			if reopening then
				return

			lastText = tb.Text
			lastCursor = tb.CursorPosition > 0 and tb.CursorPosition or (#tb.Text + 1)
		end)
		tb.FocusLost:Connect(function(enter)
			if enter then
				local txt = tb.Text
				local pos = math.clamp(lastCursor, 1, #txt + 1)
				reopening = true
				if txt == lastText then
					tb.Text = txt:sub(1, pos - 1) .. "\n" .. txt:sub(pos)
					pos += 1
				else
					pos = math.clamp(lastCursor, 1, #tb.Text + 1)
				end
				lastText = tb.Text
				task.defer(function()
					if tb.Parent then
						tb:CaptureFocus()
						pcall(function()
							tb.CursorPosition = math.clamp(pos, 1, #tb.Text + 1)
						end)
					end
					task.delay(0.15, function()
						reopening = false
					end)
				end)
				return
			end
			if reopening then
				return
			end
			obj.CurrentValue = tb.Text
			flashGlow(box, 6, 0.4)
			safe(o.Callback, tb.Text)
			autosave()
		end)
		clearBtn.Activated:Connect(function()
			tb.Text = ""
			obj.CurrentValue = ""
			flashGlow(box, 6, 0.4)
			safe(o.Callback, "")
			autosave()
		end)
		function obj:Set(t)
			tb.Text = tostring(t)
			obj.CurrentValue = tb.Text
			safe(o.Callback, tb.Text)
			autosave()
		end
		function obj:Get()
			return tb.Text
		end
		register(o.Name or "Code input", "Input", f, tab)
		flag(o, obj)
		return obj
	end

	function api:CreatePowerInput(o)
		local decimal = o.AllowDecimal == true
		local negative = o.AllowNegative == true
		local function clean(s)
			s = tostring(s)
			local out, dot = "", false
			for i = 1, #s do
				local ch = s:sub(i, i)
				if ch:match("%d") then
					out = out .. ch
				elseif ch == "." and decimal and not dot then
					dot = true
					out = out .. ch
				elseif ch == "-" and negative and #out == 0 then
					out = out .. ch
				end
			end
			return out
		end
		local function limitValue(n)
			if o.Min then
				n = math.max(n, o.Min)
			end
			if o.Max then
				n = math.min(n, o.Max)
			end
			return n
		end
		local function fmt(n)
			if n == math.floor(n) then
				return string.format("%d", n)
			end
			return tostring(n)
		end

		local f = row(page, 40, nextOrder())
		makeLabel(f, { Position = UDim2.fromOffset(12, 0), Size = UDim2.new(0.5, -12, 1, 0), Text = o.Name, Font = Enum.Font.GothamMedium, TextSize = 14 })
		local box = new("Frame", {
			AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -8, 0.5, 0), Size = UDim2.new(0.3, 0, 0, 26),
			BackgroundColor3 = C(20, 22, 28), BorderSizePixel = 0,
		}, f)
		corner(box, 6)
		metalStroke(box, 1.5)
		local startValue = limitValue(tonumber(o.CurrentValue) or o.Min or 0)
		local tb = new("TextBox", {
			Position = UDim2.fromOffset(6, 0), Size = UDim2.new(1, -12, 1, 0), BackgroundTransparency = 1,
			ClearTextOnFocus = o.ClearTextOnFocus ~= false, PlaceholderText = o.PlaceholderText or "0",
			PlaceholderColor3 = MUTED, Text = fmt(startValue), Font = Enum.Font.GothamMedium, TextSize = 12,
			TextColor3 = TEXT, TextXAlignment = Enum.TextXAlignment.Center,
		}, box)
		numericKeyboard(tb)
		tb:GetPropertyChangedSignal("Text"):Connect(function()
			local c = clean(tb.Text)
			if c ~= tb.Text then
				tb.Text = c
			end
		end)

		local obj = { Type = "PowerInput", CurrentValue = startValue, Name = o.Name }
		tb.FocusLost:Connect(function(enter)
			if enter then
				local n = tonumber(clean(tb.Text))
				if n == nil then
					tb.Text = fmt(obj.CurrentValue)
					return
				end
				n = limitValue(n)
				obj.CurrentValue = n
				tb.Text = fmt(n)
				flashGlow(box, 6, 0.4)
				overlayFlash(f, 8, Theme.Flash, 0.82, 0.6)
				safe(o.Callback, n)
				autosave()
			else
				tb.Text = fmt(obj.CurrentValue)
			end
		end)
		function obj:Set(v)
			local n = tonumber(v)
			if n == nil then
				return
			end
			n = limitValue(n)
			obj.CurrentValue = n
			tb.Text = fmt(n)
			safe(o.Callback, n)
			autosave()
		end
		register(o.Name, "Input", f, tab)
		flag(o, obj)
		return obj
	end

	function api:CreateKeybind(o)
		local f = row(page, 40, nextOrder())
		makeLabel(f, { Position = UDim2.fromOffset(12, 0), Size = UDim2.new(0.5, -12, 1, 0), Text = o.Name, Font = Enum.Font.GothamMedium, TextSize = 14 })
		local obj = { Type = "Keybind", CurrentKeybind = o.CurrentKeybind or "None", Name = o.Name }
		local btn = new("TextButton", {
			AnchorPoint = Vector2.new(1, 0.5), Position = UDim2.new(1, -8, 0.5, 0), Size = UDim2.fromOffset(96, 26),
			BackgroundColor3 = C(20, 22, 28), AutoButtonColor = false, Text = obj.CurrentKeybind,
			Font = Enum.Font.GothamMedium, TextSize = 12, TextColor3 = TEXT,
		}, f)
		corner(btn, 6)
		metalStroke(btn, 1.5)
		local listening = false
		local function setKey(name, silent)
			obj.CurrentKeybind = name
			btn.Text = name
			if not silent then
				if o.OnSet then
					pcall(o.OnSet, name)
				end
				autosave()
			end
		end
		btn.Activated:Connect(function()
			listening = true
			SpaceHub._listening = true
			btn.Text = "..."
		end)
		bind(UserInputService.InputBegan, function(input, gp)
			if listening then
				if input.UserInputType == Enum.UserInputType.Keyboard then
					listening = false
					task.defer(function()
						SpaceHub._listening = false
					end)
					if input.KeyCode == Enum.KeyCode.Escape then
						btn.Text = obj.CurrentKeybind
					else
						setKey(input.KeyCode.Name)
					end
				end
				return
			end
			if gp or SpaceHub._listening then
				return
			end
			if input.KeyCode ~= Enum.KeyCode.Unknown and input.KeyCode.Name == obj.CurrentKeybind then
				if o.HoldToInteract then
					safe(o.Callback, true)
				else
					safe(o.Callback, obj.CurrentKeybind)
				end
			end
		end)
		bind(UserInputService.InputEnded, function(input)
			if o.HoldToInteract and input.KeyCode ~= Enum.KeyCode.Unknown and input.KeyCode.Name == obj.CurrentKeybind then
				safe(o.Callback, false)
			end
		end)
		function obj:Set(name)
			setKey(tostring(name))
		end
		register(o.Name, "Keybind", f, tab)
		flag(o, obj)
		return obj
	end

		}, ex.Body)
		corner(hue, 6)
		new("UIGradient", {
			Rotation = 90,
			Color = ColorSequence.new({
				ColorSequenceKeypoint.new(0, C(255, 0, 0)), ColorSequenceKeypoint.new(0.17, C(255, 255, 0)),
				ColorSequenceKeypoint.new(0.33, C(0, 255, 0)), ColorSequenceKeypoint.new(0.5, C(0, 255, 255)),
				ColorSequenceKeypoint.new(0.67, C(0, 0, 255)), ColorSequenceKeypoint.new(0.83, C(255, 0, 255)),

		new("UIStroke", { Thickness = 1, Color = Color3.new(0, 0, 0) }, hueKnob)

			color = Color3.fromHSV(h, s, v)
			obj.Color = color
			swatch.BackgroundColor3 = color
			sv.BackgroundColor3 = Color3.fromHSV(h, 1, 1)
			svKnob.Position = UDim2.new(s, 0, 1 - v, 0)
			hueKnob.Position = UDim2.new(0.5, 0, h, 0)
			if fire then
				safe(o.Callback, color)
				autosave()
			end
		end
		refresh(false)

		local mode
		local function drag(pos)
			if mode == "sv" then
				s = math.clamp((pos.X - sv.AbsolutePosition.X) / math.max(sv.AbsoluteSize.X, 1), 0, 1)
				v = 1 - math.clamp((pos.Y - sv.AbsolutePosition.Y) / math.max(sv.AbsoluteSize.Y, 1), 0, 1)
			elseif mode == "hue" then
				h = math.clamp((pos.Y - hue.AbsolutePosition.Y) / math.max(hue.AbsoluteSize.Y, 1), 0, 1)
			else
				return
			end
			refresh(true)
		end
		local function start(m)
			return function(input)
				if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
					mode = m
					page.ScrollingEnabled = false
					drag(screenPos(input))
				end
			end
		end
		sv.InputBegan:Connect(start("sv"))
		hue.InputBegan:Connect(start("hue"))
		bind(UserInputService.InputChanged, function(input)
			if mode and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
				drag(screenPos(input))
			end
		end)
		bind(UserInputService.InputEnded, function(input)
			if mode and (input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch) then
				mode = nil
				page.ScrollingEnabled = true
			end
		end)
		function obj:Set(c)
			h, s, v = c:ToHSV()
			refresh(true)
		end
		register(o.Name, "ColorPicker", ex.Head, tab)
		flag(o, obj)
		return obj
	end

	function api:CreateCodeBlock(o)
		local height = o.Height or 240
		local f = row(page, height, nextOrder())
		makeLabel(f, {
			Position = UDim2.fromOffset(12, 0), Size = UDim2.new(1, -110, 0, 34),
			Text = o.Title or "Code", Font = Enum.Font.GothamMedium, TextSize = 13,
			TextTruncate = Enum.TextTruncate.AtEnd,
		})
		local copyFrame, copyBtn = makeButton(f, "Copy", 1, 24)
		copyFrame.AnchorPoint = Vector2.new(1, 0)
		copyFrame.Position = UDim2.new(1, -8, 0, 5)
		copyFrame.Size = UDim2.fromOffset(78, 24)
		copyBtn.TextSize = 12

		local box = new("Frame", {
			Position = UDim2.fromOffset(8, 34), Size = UDim2.new(1, -16, 1, -42),
			BackgroundColor3 = C(14, 16, 22), BorderSizePixel = 0,
		}, f)
		corner(box, 6)
		metalStroke(box, 1)
		local scroll = new("ScrollingFrame", {
			Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, BorderSizePixel = 0, ScrollBarThickness = 4,
			CanvasSize = UDim2.new(), AutomaticCanvasSize = Enum.AutomaticSize.XY,
			ScrollingDirection = Enum.ScrollingDirection.XY,
		}, box)
		themed(scroll, "ScrollBarImageColor3", "Accent")
		pad(scroll, 6, 8, 6, 8)
		local list = new("Frame", {
			Size = UDim2.new(0, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.XY, BackgroundTransparency = 1,
		}, scroll)
		new("UIListLayout", { SortOrder = Enum.SortOrder.LayoutOrder }, list)

		local obj = { Type = "CodeBlock", CurrentValue = o.Code or "", Name = o.Title or "Code" }
		local function render(code)
			for _, c in ipairs(list:GetChildren()) do
				if c:IsA("TextLabel") then
					c:Destroy()
				end
			end
			local lines = string.split((tostring(code):gsub("\r", "")), "\n")
			local chunk, n, idx = {}, 0, 0
			local function flush()
				if #chunk == 0 then
					return
				end
				idx += 1
				new("TextLabel", {
					LayoutOrder = idx, Size = UDim2.new(0, 0, 0, 0), AutomaticSize = Enum.AutomaticSize.XY,
					BackgroundTransparency = 1, Font = Enum.Font.Code, TextSize = 12, TextColor3 = TEXT,
					TextXAlignment = Enum.TextXAlignment.Left, TextYAlignment = Enum.TextYAlignment.Top,
					TextWrapped = false, RichText = false, Text = table.concat(chunk, "\n"),
				}, list)
				chunk, n = {}, 0
			end
			for _, line in ipairs(lines) do
				n += 1
				chunk[n] = (line:gsub("\t", "    "))
				if n >= 60 then
					flush()
				end
			end
			flush()
		end
		render(obj.CurrentValue)

		local busy = false
		copyBtn.Activated:Connect(function()
			if busy then
				return
			end
			busy = true
			local ok = copyText(obj.CurrentValue)
			copyBtn.Text = ok and "Copied!" or "Unavailable"
			copyBtn.TextColor3 = ok and GREEN or RED
			task.delay(1.2, function()
				copyBtn.Text = "Copy"
				copyBtn.TextColor3 = TEXT
				busy = false
			end)
		end)
		function obj:Set(code)
			obj.CurrentValue = tostring(code)
			render(obj.CurrentValue)
		end
		register(o.Title or "Code", "Label", f, tab)
		return obj
	end

	function api:CreateFloatingButtons(o)
		o = o or {}
		local maxCount = o.Max or 40
		local count = math.clamp(math.floor(o.Count or 5), 1, maxCount)
		local names = o.Names or {}
		local size = math.floor((o.Size or 56) * deviceScale + 0.5)
		local fgui
		local conns = {}

		local function destroyButtons()
			for _, c in ipairs(conns) do
				pcall(function()
					c:Disconnect()
				end)
			end
			table.clear(conns)
			if fgui then
				fgui:Destroy()
				fgui = nil
			end
		end

		local function build()
			if fgui and fgui.Parent then
				return
			end
			fgui = createGui("SpaceHubFloating_" .. tostring(o.Flag or o.Name or "Buttons"))
			local vp = workspace.CurrentCamera.ViewportSize
			local perRow = math.max(1, math.floor((vp.X - 40) / (size + 10)))
			perRow = math.min(perRow, 5)

			for i = 1, count do
				local col, rowN = (i - 1) % perRow, math.floor((i - 1) / perRow)
				local frame = new("Frame", {
					Position = UDim2.fromOffset(20 + col * (size + 10), vp.Y * 0.35 + rowN * (size + 10)),
					Size = UDim2.fromOffset(size, size), BackgroundColor3 = WHITE, BorderSizePixel = 0, Active = true,
				}, fgui)
				corner(frame, 12)
				new("UIGradient", { Rotation = 90, Color = ColorSequence.new(STEEL_A, STEEL_B) }, frame)
				metalStroke(frame, 2)
				makeLabel(frame, {
					Size = UDim2.fromScale(1, 1), Text = tostring(names[i] or i), Font = Enum.Font.GothamBold,
					TextSize = 16, TextXAlignment = Enum.TextXAlignment.Center, TextWrapped = true,
				})
				local hit = new("TextButton", {
					Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, Text = "", ZIndex = 5,
				}, frame)

				local press
				hit.InputBegan:Connect(function(input)
					local t = input.UserInputType
					if t == Enum.UserInputType.MouseButton1 or t == Enum.UserInputType.Touch then
						press = {
							start = input.Position, kind = t, moved = false,
							pos = Vector2.new(frame.Position.X.Offset, frame.Position.Y.Offset),
						}
					end
				end)
				table.insert(conns, UserInputService.InputChanged:Connect(function(input)
					if not press then
						return
					end
					if input.UserInputType ~= Enum.UserInputType.MouseMovement and input.UserInputType ~= Enum.UserInputType.Touch then
						return
					end
					local d = input.Position - press.start
					if not press.moved and d.Magnitude > 8 then
						press.moved = true
					end
					if press.moved then
						local v = workspace.CurrentCamera.ViewportSize
						frame.Position = UDim2.fromOffset(
							math.clamp(press.pos.X + d.X, 0, v.X - size),
							math.clamp(press.pos.Y + d.Y, 0, v.Y - size)
						)
					end
				end))
				table.insert(conns, UserInputService.InputEnded:Connect(function(input)
					if press and input.UserInputType == press.kind then
						local p = press
						press = nil
						if not p.moved then
							flashGlow(frame, 12, 0.5)
							safe(o.Callback, i, names[i] or i)
						end
					end
				end))
			end
			fgui.Destroying:Connect(function()
				for _, c in ipairs(conns) do
					pcall(function()
						c:Disconnect()
					end)
				end
			end)
		end

		local obj = api:CreateToggle({
			Name = o.Name or "button float",
			CurrentValue = o.CurrentValue == true,
			Flag = o.Flag,
			Callback = function(on)
				if on then
					build()
				else
					destroyButtons()
				end
				safe(o.OnToggle, on)
			end,
		})
		obj.Type = "Toggle"
		if obj.CurrentValue then
			build()
		end
		function obj:SetCount(n)
			count = math.clamp(math.floor(tonumber(n) or count), 1, maxCount)
			if obj.CurrentValue then
				destroyButtons()
				build()
			end
		end
		local amount = api:CreatePowerInput({
			Name = o.AmountName or "x amount",
			CurrentValue = count, Min = 1, Max = maxCount, PlaceholderText = "5",
			Flag = o.Flag and (o.Flag .. "_Amount") or nil,
			Callback = function(n)
				obj:SetCount(n)
			end,
		})
		obj.Amount = amount
		return obj
	end

	function api:CreateSpectate(o)
		o = o or {}
		local cam = workspace.CurrentCamera
		local f = row(page, 172, nextOrder())

		local function arrowButton(text, left)
			local fr, b = makeButton(f, text, 1, 44)
			fr.AnchorPoint = Vector2.new(left and 0 or 1, 0.5)
			fr.Position = UDim2.new(left and 0 or 1, left and 12 or -12, 0, 64)
			fr.Size = UDim2.fromOffset(44, 44)
			b.TextSize = 22
			b.Font = Enum.Font.GothamBold
			return b
		end
		local prevBtn = arrowButton("<", true)
		local nextBtn = arrowButton(">", false)

		local ring = new("Frame", {
			AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.new(0.5, 0, 0, 64), Size = UDim2.fromOffset(92, 92),
			BackgroundColor3 = GREEN, BackgroundTransparency = 1, BorderSizePixel = 0, ZIndex = 1,
		}, f)
		corner(ring, UDim.new(1, 0))
		local photo = new("ImageLabel", {
			AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.new(0.5, 0, 0, 64), Size = UDim2.fromOffset(84, 84),
			BackgroundColor3 = C(20, 22, 28), BorderSizePixel = 0, ZIndex = 2,
		}, f)
		corner(photo, UDim.new(1, 0))
		metalStroke(photo, 2)
		local photoHit = new("TextButton", {
			AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.new(0.5, 0, 0, 64), Size = UDim2.fromOffset(84, 84),
			BackgroundTransparency = 1, Text = "", ZIndex = 3,
		}, f)
		corner(photoHit, UDim.new(1, 0))

		local nameLbl = makeLabel(f, {
			Position = UDim2.fromOffset(12, 116), Size = UDim2.new(1, -24, 0, 22), Text = "", Font = Enum.Font.GothamBold,
			TextSize = 14, TextXAlignment = Enum.TextXAlignment.Center, TextTruncate = Enum.TextTruncate.AtEnd,
		})
		local subLbl = makeLabel(f, {
			Position = UDim2.fromOffset(12, 138), Size = UDim2.new(1, -24, 0, 18), Text = "", TextColor3 = MUTED,
			TextSize = 11, TextXAlignment = Enum.TextXAlignment.Center, TextTruncate = Enum.TextTruncate.AtEnd,
		})

		local list, idx, spectating = {}, 1, false
		local thumbs = {}
		local obj = { Type = "Spectate", CurrentValue = false, Name = o.Name or "Spectate" }

		local function current()
			return list[idx]
		end
		local function refreshList()
			local cur = current()
			list = {}
			for _, p in ipairs(Players:GetPlayers()) do
				if p ~= player then
					table.insert(list, p)
				end
			end
			table.sort(list, function(a, b)
				return a.Name:lower() < b.Name:lower()
			end)
			idx = (cur and table.find(list, cur)) or math.clamp(idx, 1, math.max(#list, 1))
		end
		local function showCurrent()
			local p = current()
			if not p then
				nameLbl.Text = "No players"
				subLbl.Text = "Waiting for players..."
				photo.Image = ""
				return
			end
			nameLbl.Text = p.DisplayName
			subLbl.Text = "@" .. p.Name .. (spectating and "  -  spectating" or "  -  tap photo to spectate")
			if thumbs[p.UserId] then
				photo.Image = thumbs[p.UserId]
			else
				photo.Image = ""
				task.spawn(function()
					local ok, img = pcall(function()
						return Players:GetUserThumbnailAsync(p.UserId, Enum.ThumbnailType.HeadShot, Enum.ThumbnailSize.Size150x150)
					end)
					if ok and img then
						thumbs[p.UserId] = img
						if current() == p then
							photo.Image = img
						end
					end
				end)
			end
		end

		local function resetCamera()
			local ch = player.Character
			local hum = ch and ch:FindFirstChildOfClass("Humanoid")
			if hum then
				cam.CameraSubject = hum
			end
		end
		local function stop()
			if not spectating then
				return
			end
			spectating = false
			obj.CurrentValue = false
			tw(ring, 0.25, { BackgroundTransparency = 1 })
			resetCamera()
			showCurrent()
			safe(o.OnStop)
			safe(o.Callback, current(), false)
		end
		local function start()
			local p = current()
			if not p or spectating then
				return
			end
			spectating = true
			obj.CurrentValue = true
			tw(ring, 0.25, { BackgroundTransparency = 0.15 })
			showCurrent()
			safe(o.Callback, p, true)
			task.spawn(function()
				while spectating and f.Parent do
					local t = current()
					local hum = t and t.Character and t.Character:FindFirstChildOfClass("Humanoid")
					if hum and cam.CameraSubject ~= hum then
						cam.CameraSubject = hum
					end
					task.wait(0.25)
				end
			end)
		end

		local function step(dir)
			if #list == 0 then
				return
			end
			idx = (idx - 1 + dir) % #list + 1
			haptic()
			showCurrent()
		end
		prevBtn.Activated:Connect(function()
			step(-1)
		end)
		nextBtn.Activated:Connect(function()
			step(1)
		end)
		photoHit.Activated:Connect(function()
			haptic()
			if spectating then
				stop()
			else
				start()
			end
		end)

		local function onPlayersChanged()
			local before = current()
			refreshList()
			if spectating and (#list == 0 or before ~= current() and not table.find(list, before)) then
				if #list == 0 then
					stop()
				end
			end
			showCurrent()
		end
		bind(Players.PlayerAdded, function()
			task.defer(onPlayersChanged)
		end)
		bind(Players.PlayerRemoving, function()
			task.defer(onPlayersChanged)
		end)

		function obj:Next()
			step(1)
		end
		function obj:Previous()
			step(-1)
		end
		function obj:Set(v)
			if v then
				start()
			else
				stop()
			end
		end
		function obj:Refresh()
			onPlayersChanged()
		end

		refreshList()
		showCurrent()
		register(o.Name or "Spectate", "Spectate", f, tab)
		return obj
	end

	local extCtx = { Api = api, Page = page, Tab = tab, Register = register, Flag = flag, NextOrder = nextOrder }
	for extName, ext in pairs(SpaceHub._extensions or {}) do
		local original = api[extName]
		if original == nil or ext.override then
			local c = setmetatable({ Original = original }, { __index = extCtx })
			api[extName] = function(_, o)
				return ext.fn(c, o)
			end
		end
	end

	return api
end
